cmake_minimum_required(VERSION 3.24)
project(SweetDisplayFoundations LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This foundation target uses Windows Media Foundation.")
endif()
add_executable(SweetDisplayEncoderProbe windows/tools/EncoderProbe.cpp)
target_compile_features(SweetDisplayEncoderProbe PRIVATE cxx_std_17)
target_compile_definitions(SweetDisplayEncoderProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(SweetDisplayEncoderProbe PRIVATE mfplat mfuuid ole32)
if(MSVC)
  target_compile_options(SweetDisplayEncoderProbe PRIVATE /W4 /WX /utf-8)
endif()
